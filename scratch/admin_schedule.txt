'use client';

import {
    useState,
    useEffect,
    useRef,
} from 'react';

import { useRouter } from 'next/navigation';
import Link from 'next/link';
import {
    Plus,
    ChevronLeft,
    ChevronRight,
} from 'lucide-react';

import SessionSlot from '@/components/admin/SessionSlot';
import { toggleSlotAction } from './actions';

import type {
    AdminScheduleSlot,
} from '@/modules/bookings/booking-types';

const MONTHS = ['Januari', 'Februari', 'Maret', 'April', 'Mei', 'Juni', 'Juli', 'Agustus', 'September', 'Oktober', 'November', 'Desember'];
const DAYS = ['MIN', 'SEN', 'SEL', 'RAB', 'KAM', 'JUM', 'SAB'];

function getDaysInMonth(year: number, month: number) {
    const date = new Date(year, month, 1);
    const days = [];
    while (date.getMonth() === month) {
        days.push(new Date(date));
        date.setDate(date.getDate() + 1);
    }
    return days;
}

function TimelineSlot({
    time,
    slot,
}: {
    time: string;
    slot: AdminScheduleSlot | undefined;
}) {
    return (
        <div style={{ display: 'flex', minHeight: '80px' }}>
            <div
                style={{
                    width: '45px',
                    flexShrink: 0,
                    textAlign: 'right',
                    paddingRight: '12px',
                    paddingTop: '4px',
                }}
            >
                <div
                    style={{
                        fontSize: '15px',
                        fontFamily: 'var(--font-display)',
                        fontWeight: 500,
                        color: 'var(--ink)',
                    }}
                >
                    {time}
                </div>

                <div
                    style={{
                        fontSize: '9px',
                        fontWeight: 700,
                        color: 'var(--muted)',
                        marginTop: '2px',
                    }}
                >
                    {slot
                        ? `${slot.booking?.participants.length ?? 0}/${slot.capacity}`
                        : '—'}
                </div>
            </div>

            <div
                style={{
                    display: 'flex',
                    flexDirection: 'column',
                    alignItems: 'center',
                }}
            >
                <div
                    style={{
                        width: '14px',
                        height: '14px',
                        borderRadius: '50%',
                        border: `2.5px solid ${slot?.booking
                            ? 'var(--ink)'
                            : 'rgba(0,0,0,0.2)'
                            }`,
                        backgroundColor: '#F9F9F8',
                        flexShrink: 0,
                        marginTop: '7px',
                    }}
                />

                <div
                    style={{
                        width: '1.5px',
                        flexGrow: 1,
                        backgroundColor: 'rgba(0,0,0,0.08)',
                        margin: '6px 0',
                    }}
                />
            </div>

            <div
                style={{
                    flex: 1,
                    paddingLeft: '16px',
                    paddingBottom: '24px',
                }}
            >
                {slot?.slotExists ? (
                    <SessionSlot slot={slot} />
                ) : (
                    <div
                        onClick={async () => {
                            if (confirm(`Buka slot ini (${slot?.startTime}) untuk di-booking?`)) {
                                const res = await toggleSlotAction(
                                    slot!.date,
                                    slot!.startTime,
                                    slot!.endTime,
                                    slot!.capacity,
                                    true
                                );
                                if (!res.ok) {
                                    alert(res.message);
                                } else {
                                    window.location.reload();
                                }
                            }
                        }}
                        style={{
                            border: '1.5px dashed rgba(0,0,0,0.15)',
                            borderRadius: '20px',
                            padding: '16px',
                            display: 'flex',
                            alignItems: 'center',
                            justifyContent: 'center',
                            gap: '8px',
                            color: 'var(--muted)',
                            background: 'rgba(0,0,0,0.02)',
                            cursor: 'pointer'
                        }}
                    >
                        <Plus size={16} strokeWidth={3} />
                        <span
                            style={{
                                fontSize: '13px',
                                fontWeight: 600,
                            }}
                        >
                            Buka Slot
                        </span>
                    </div>
                )}
            </div>
        </div>
    );
}

export default function AdminScheduleClient({
    initialSlots,
    initialDate,
}: {
    initialSlots: AdminScheduleSlot[];
    initialDate: string;
}) {
    const [activeChip, setActiveChip] = useState('Semua');

    const parsedInitialDate = new Date(`${initialDate}T00:00:00`);

    const [currentDate, setCurrentDate] = useState(
        new Date(
            parsedInitialDate.getFullYear(),
            parsedInitialDate.getMonth(),
            1
        )
    );

    const [selectedDate, setSelectedDate] = useState(
        parsedInitialDate
    );

    const daysInMonth = getDaysInMonth(currentDate.getFullYear(), currentDate.getMonth());
    const scrollRef = useRef<HTMLDivElement>(null);
    const router = useRouter();

    const morningSlots = initialSlots.filter(
        (slot) => slot.block === 'MORNING'
    );

    const afternoonSlots = initialSlots.filter(
        (slot) => slot.block === 'AFTERNOON'
    );

    const totalCapacity = initialSlots.reduce(
        (total, slot) => total + slot.capacity,
        0
    );

    const occupiedSeats = initialSlots.reduce(
        (total, slot) =>
            total + (slot.booking?.participants.length ?? 0),
        0
    );

    const activeSessions = initialSlots.filter(
        (slot) => slot.slotExists && slot.isOpen
    ).length;

    const emptySlots = initialSlots.filter(
        (slot) => !slot.slotExists
    ).length;

    const closedSlots = initialSlots.filter(
        (slot) => slot.slotExists && !slot.isOpen
    ).length;

    const occupancyPercentage =
        totalCapacity > 0
            ? Math.round((occupiedSeats / totalCapacity) * 100)
            : 0;

    const chips = [
        {
            label: 'Semua',
            count: null,
        },
        {
            label: 'Slot kosong',
            count: emptySlots,
        },
        {
            label: 'Pagi',
            count: morningSlots.length,
        },
        {
            label: 'Sore',
            count: afternoonSlots.length,
        },
    ];

    const visibleMorningSlots =
        activeChip === 'Slot kosong'
            ? morningSlots.filter((slot) => !slot.slotExists)
            : morningSlots;

    const visibleAfternoonSlots =
        activeChip === 'Slot kosong'
            ? afternoonSlots.filter((slot) => !slot.slotExists)
            : afternoonSlots;

    // Auto-scroll to selected date on mount or month change
    useEffect(() => {
        if (scrollRef.current) {
            const selectedEl = scrollRef.current.querySelector('.selected-day') as HTMLElement;
            if (selectedEl) {
                scrollRef.current.scrollTo({
                    left: selectedEl.offsetLeft - scrollRef.current.offsetWidth / 2 + selectedEl.offsetWidth / 2,
                    behavior: 'smooth'
                });
            }
        }
    }, [currentDate.getMonth()]);

    const handlePrevMonth = () => {
        setCurrentDate(new Date(currentDate.getFullYear(), currentDate.getMonth() - 1, 1));
    };

    const handleNextMonth = () => {
        setCurrentDate(new Date(currentDate.getFullYear(), currentDate.getMonth() + 1, 1));
    };

    const handleSelectDate = (date: Date) => {
        setSelectedDate(date);

        if (date.getMonth() !== currentDate.getMonth()) {
            setCurrentDate(
                new Date(date.getFullYear(), date.getMonth(), 1)
            );
        }

        const dateParam = [
            date.getFullYear(),
            String(date.getMonth() + 1).padStart(2, '0'),
            String(date.getDate()).padStart(2, '0'),
        ].join('-');

        router.push(`/admin/schedule?date=${dateParam}`);
    };

    return (
        <section className="screen active" data-page="schedule">
            {/* 1. Appbar */}
            <div className="appbar" style={{ paddingBottom: '16px', display: 'flex', alignItems: 'center', justifyContent: 'space-between' }}>
                <Link href="/admin" style={{ width: '36px', height: '36px', display: 'flex', alignItems: 'center', justifyContent: 'center', borderRadius: '50%', border: '1px solid rgba(0,0,0,0.1)', background: '#fff', color: 'var(--ink)' }}>
                    <ChevronLeft size={18} strokeWidth={2.5} />
                </Link>
                <span style={{ fontSize: '15px', fontWeight: 700 }}>Jadwal Sesi</span>
                <button style={{ width: '36px', height: '36px', display: 'flex', alignItems: 'center', justifyContent: 'center', borderRadius: '50%', background: 'var(--ink)', color: '#fff', border: 'none', cursor: 'pointer' }} onClick={() => alert('Add session')}>
                    <Plus size={18} strokeWidth={2.5} />
                </button>
            </div>

            {/* 2. Dynamic Month & Date Picker */}
            <div style={{ display: 'flex', alignItems: 'center', justifyContent: 'space-between', marginBottom: '16px', marginTop: '8px', padding: '0 4px' }}>
                <button onClick={handlePrevMonth} style={{ padding: '8px', marginLeft: '-8px', color: 'var(--muted)', background: 'transparent', border: 'none' }}>
                    <ChevronLeft size={18} />
                </button>
                <span style={{ fontSize: '15px', fontWeight: 700, color: 'var(--ink)' }}>
                    {MONTHS[currentDate.getMonth()]} {currentDate.getFullYear()}
                </span>
                <button onClick={handleNextMonth} style={{ padding: '8px', marginRight: '-8px', color: 'var(--muted)', background: 'transparent', border: 'none' }}>
                    <ChevronRight size={18} />
                </button>
            </div>

            <div
                ref={scrollRef}
                className="daystrip"
                style={{ display: 'flex', gap: '8px', overflowX: 'auto', paddingBottom: '16px', paddingLeft: '1px', WebkitOverflowScrolling: 'touch', scrollbarWidth: 'none', msOverflowStyle: 'none', scrollSnapType: 'x mandatory' }}
            >
                {daysInMonth.map((date) => {
                    const isSelected = date.getDate() === selectedDate.getDate() && date.getMonth() === selectedDate.getMonth();
                    const dayName = DAYS[date.getDay()];
                    const isWeekend = date.getDay() === 0 || date.getDay() === 6;

                    return (
                        <div
                            key={date.toISOString()}
                            onClick={() => handleSelectDate(date)}
                            className={isSelected ? 'selected-day' : ''}
                            style={{
                                scrollSnapAlign: 'center', flexShrink: 0, width: '48px', height: '68px', borderRadius: '18px', border: '1px solid', borderColor: isSelected ? 'var(--ink)' : 'rgba(0,0,0,0.05)', display: 'flex', flexDirection: 'column', alignItems: 'center', justifyContent: 'center', cursor: 'pointer', transition: '0.2s',
                                background: isSelected ? 'var(--ink)' : '#fff', color: isSelected ? '#fff' : 'var(--ink)', boxShadow: isSelected ? '0 4px 12px rgba(0,0,0,0.1)' : 'none'
                            }}
                        >
                            <span style={{ fontSize: '9px', fontWeight: 700, opacity: isSelected ? 0.8 : 0.4 }}>
                                {dayName}
                            </span>
                            <b style={{ fontSize: '19px', marginTop: '2px' }}>{date.getDate()}</b>
                            <i style={{
                                width: '4px', height: '4px', borderRadius: '50%', marginTop: '4px',
                                background: (isWeekend && !isSelected) ? '#C99A4B' : 'currentColor',
                                opacity: (!isSelected && !isWeekend) ? 0.25 : 1
                            }} />
                        </div>
                    );
                })}
            </div>

            {/* Stats Card */}
            <div className="card" style={{ marginTop: '8px' }}>
                <div className="cardhead">
                    <b>Kapasitas {DAYS[selectedDate.getDay()]}, {selectedDate.getDate()} {MONTHS[selectedDate.getMonth()].substring(0, 3)}</b>
                    <span className="pill ok">Sehat</span>
                </div>
                <div className="meter">
                    <i
                        style={{
                            width: `${occupancyPercentage}%`,
                        }}
                    />
                </div>

                <div className="meterlabel">
                    <span>
                        {occupiedSeats} dari {totalCapacity} Slot Terisi
                    </span>

                    <span>
                        {occupancyPercentage}%
                    </span>
                </div>

                <div
                    style={{
                        display: 'flex',
                        gap: '16px',
                        marginTop: '13px',
                        fontSize: '11px',
                        color: 'var(--muted)',
                    }}
                >
                    <span>
                        ● {activeSessions} Sesi Aktif
                    </span>

                    <span>
                        ● {emptySlots} Slot Kosong
                    </span>

                    <span>
                        ● {closedSlots} Ditutup
                    </span>
                </div>
            </div>

            <div className="chips" data-chipgroup style={{ display: 'flex', gap: '8px', overflowX: 'auto', paddingBottom: '4px', WebkitOverflowScrolling: 'touch', marginTop: '24px' }}>

                {chips.map((chip) => (
                    <button
                        key={chip.label}
                        className={`chip ${activeChip === chip.label ? 'active' : ''
                            }`}
                        onClick={() => setActiveChip(chip.label)}
                        style={{
                            whiteSpace: 'nowrap',
                        }}
                    >
                        {chip.label}
                        {chip.count !== null && ` · ${chip.count}`}
                    </button>
                ))}
            </div>

            {/* 3. Timeline Revamp with Flexbox (Inline Styles) */}
            <div style={{ marginTop: '24px', paddingBottom: '96px' }}>
                {(
                    activeChip === 'Semua' ||
                    activeChip === 'Pagi' ||
                    activeChip === 'Slot kosong'
                ) && visibleMorningSlots.length > 0 && (
                        <div style={{ marginBottom: '32px' }}>
                            <div
                                style={{
                                    display: 'flex',
                                    alignItems: 'center',
                                    justifyContent: 'space-between',
                                    marginBottom: '20px',
                                    padding: '0 4px',
                                }}
                            >
                                <b style={{ fontSize: '17px' }}>Pagi</b>

                                <span
                                    style={{
                                        fontSize: '12px',
                                        color: 'var(--muted)',
                                        fontWeight: 500,
                                    }}
                                >
                                    {morningSlots.length > 0 && (
                                        <>
                                            {morningSlots[0].startTime} —{' '}
                                            {morningSlots[morningSlots.length - 1].startTime}
                                        </>
                                    )}
                                </span>
                            </div>

                            <div style={{ display: 'flex', flexDirection: 'column' }}>
                                {visibleMorningSlots.map((slot) => (
                                    <TimelineSlot
                                        key={`${slot.block}-${slot.startTime}`}
                                        time={slot.startTime}
                                        slot={slot}
                                    />
                                ))}
                            </div>
                        </div>
                    )}

                {(
                    activeChip === 'Semua' ||
                    activeChip === 'Sore' ||
                    activeChip === 'Slot kosong'
                ) && visibleAfternoonSlots.length > 0 && (
                        <div>
                            <div
                                style={{
                                    display: 'flex',
                                    alignItems: 'center',
                                    justifyContent: 'space-between',
                                    marginBottom: '20px',
                                    padding: '0 4px',
                                }}
                            >
                                <b style={{ fontSize: '17px' }}>Sore</b>

                                <span
                                    style={{
                                        fontSize: '12px',
                                        color: 'var(--muted)',
                                        fontWeight: 500,
                                    }}
                                >
                                    {afternoonSlots.length > 0 && (
                                        <>
                                            {afternoonSlots[0].startTime} —{' '}
                                            {afternoonSlots[afternoonSlots.length - 1].startTime}
                                        </>
                                    )}
                                </span>
                            </div>

                            <div style={{ display: 'flex', flexDirection: 'column' }}>
                                {visibleAfternoonSlots.map((slot) => (
                                    <TimelineSlot
                                        key={`${slot.block}-${slot.startTime}`}
                                        time={slot.startTime}
                                        slot={slot}
                                    />
                                ))}
                            </div>
                        </div>
                    )}
            </div>
        </section>
    );
}
